#
# The function accepts following parameters:
#  1. INTEGER n
#  2. INTEGER_ARRAY arr
#

def insertionSort1(n, arr):
    temp = arr[n-1]
    
    for j in range(n-2, -1, -1):
        if arr[j] > temp:
            arr[j+1] = arr[j]
            print(" ".join(map(str, arr)))
        else:
            arr[j+1] = temp
            print(" ".join(map(str, arr)))
            return
    
    
    arr[0] = temp
    print(" ".join(map(str, arr)))

if __name__ == '__main__':
    n = int(input().strip())

    arr = list(map(int, input().rstrip().split()))

    insertionSort1(n, arr)
