class Solution:
    def dailyTemperatures(self, temperatures: List[int]) -> List[int]:
        result = [0]* len(temperatures)
        stack = []
        for i , tmep in enumerate(temperatures):
            while stack and temperatures[stack[-1]] < tmep:
                index = stack.pop()
                result[index] = i - index
            stack.append(i)
        return result
