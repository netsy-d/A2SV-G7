class Solution:
    def appendCharacters(self, s: str, t: str) -> int:
        if t[0] not in s:
            return len(t)
        else:
            i= j =0
            count = 0
            while i < len(s) and j < len(t):
                if s[i] == t[j]:
                    count+=1
                    j+=1
                i+=1
            return len(t) - count
