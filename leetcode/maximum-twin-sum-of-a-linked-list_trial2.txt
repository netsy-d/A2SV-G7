# Definition for singly-linked list.
# class ListNode:
#     def __init__(self, val=0, next=None):
#         self.val = val
#         self.next = next
class Solution:
    def pairSum(self, head: Optional[ListNode]) -> int:
        values = []
        while head:
            values.append(head.val)
            head = head.next
        n = len(values)
        l = 0
        r = n-1
        max_sum = 0
        while l < r:
            sum = values[l] + values[r]
            max_sum = max(max_sum,sum)
            l+=1
            r-=1
        return max_sum
