class Solution:
    def dividePlayers(self, ss: List[int]) -> int:
        
        ss.sort()
        chemistry, s = 0, ss[0] + ss[-1]

        for i in range(len(ss)//2):
            if ss[i] + ss[-i-1] == s:
                chemistry += ss[i] * ss[-i-1]
            else:
                return -1
        return chemistry
