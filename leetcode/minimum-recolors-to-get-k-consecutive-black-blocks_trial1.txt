class Solution:
    def minimumRecolors(self, blocks: str, k: int) -> int:
        min_op = float('inf')
        i = j=0
        n = len(blocks)
        
        while i <= n-k:
            count = 0
            for j in range(i,i+k):
                if blocks[j] == "W":
                    count+=1
            min_op = min(min_op,count)
            i+=1
        return min_op
